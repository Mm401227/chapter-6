 function showIOSAlert({ title = '提示', message = '', hasInput = false, defaultValue = '', buttons = [] }) {
   return new Promise((resolve) => {
     const overlay = document.getElementById('ios-modal');
     const titleEl = document.getElementById('ios-modal-title');
     const msgEl = document.getElementById('ios-modal-msg');
     const inputEl = document.getElementById('ios-modal-input');
     const btnsEl = document.getElementById('ios-modal-btns');
     titleEl.innerText = title;
     msgEl.innerText = message;  
     if (hasInput) {
       inputEl.style.display = 'block';
       inputEl.value = defaultValue;
       setTimeout(() => inputEl.focus(), 50);
     } else {
       inputEl.style.display = 'none';
     }
     btnsEl.innerHTML = '';
     buttons.forEach(btn => {
       const btnEl = document.createElement('button');
       btnEl.className = `ios-modal-btn ${btn.style || ''}`;
       btnEl.innerText = btn.text;
       btnEl.onclick = () => {
         overlay.classList.remove('active');
         resolve({ value: btn.value, inputValue: inputEl.value.trim() });
       };
       btnsEl.appendChild(btnEl);
     });
     overlay.classList.add('active');
   });
 }

   renderPalette();

 function renderPalette() {
   paletteList.innerHTML = '';
   TANGRAM_DEFINITIONS.forEach(def => {
     const item = document.createElement('div');
     item.className = 'piece-item';
     item.setAttribute('draggable', 'true');
     
     const pointsStr = def.rawPoints.map(p => `${p.x},${p.y}`).join(' ');
     item.innerHTML = `
       <svg width="64" height="64" viewBox="-10 -10 220 220">
         <polygon points="${pointsStr}" fill="${def.color}" stroke="#64748b" stroke-width="2" stroke-linejoin="round"/>
       </svg>
       <span class="piece-label">${def.name}</span>
     `;
     item.addEventListener('dragstart', (e) => {
       e.dataTransfer.setData('text/plain', def.type);
       e.dataTransfer.effectAllowed = 'copy';
     });
     item.addEventListener('click', (e) => {
       e.preventDefault();
       const rect = svgCanvas.getBoundingClientRect();
       const cx = rect.width / 2 || 300;
       const cy = rect.height / 2 || 200;
       const pieceData = createPieceData(def.type, cx, cy);
       if (pieceData) {
         const currentCanvas = getCurrentCanvas();
         currentCanvas.pieces.push(pieceData);
         renderPieceSVG(pieceData);
         highlightSelectedPiece(pieceData.id);
       }
     });
     paletteList.appendChild(item);
   });
 }

 function renderTabs() {
   const existingTabs = tabsBar.querySelectorAll('.tab-item');
   existingTabs.forEach(t => t.remove());
   canvases.forEach(c => {
     const tab = document.createElement('div');
     tab.className = `tab-item ${c.id === activeCanvasId ? 'active' : ''}`;
     tab.innerText = c.name;
     tab.title = "单击切换，双击重命名";
     
     tab.addEventListener('click', () => switchCanvas(c.id));
     tab.addEventListener('dblclick', (e) => {
       e.stopPropagation();
       renameCanvas(c.id);
     });
     tabsBar.insertBefore(tab, btnAddCanvas);
   });
 }

 async function renameCanvas(canvasId) {
   const targetCanvas = canvases.find(c => c.id === canvasId);
   if (!targetCanvas) return;
   const result = await showIOSAlert({
     title: '重命名画布',
     message: '请输入新的画布名称：',
     hasInput: true,
     defaultValue: targetCanvas.name,
     buttons: [
       { text: '取消', style: '', value: false },
       { text: '保存', style: 'primary', value: true }
     ]
   });
   if (result.value && result.inputValue) {
     targetCanvas.name = result.inputValue;
     renderTabs();
   }
 }

 function switchCanvas(id) {
   if (activeCanvasId === id) return;
   highlightSelectedPiece(null);
   activeCanvasId = id;
   renderTabs();
   renderCurrentCanvasPieces();
 }

 function createNewCanvas() {
   canvasIdCounter++;
   const newCanvas = {
     id: canvasIdCounter,
     name: `Sheet ${canvasIdCounter}`,
     pieces: []
   };
   canvases.push(newCanvas);
   switchCanvas(newCanvas.id);
 }

 async function deleteCurrentCanvas() {
   if (canvases.length <= 1) {
     showIOSAlert({
       title: '无法删除',
       message: '至少需要保留一个画布！',
       buttons: [{ text: '好的', style: 'primary' }]
     });
     return;
   }
   const res = await showIOSAlert({
     title: '删除画布',
     message: `确定要删除当前【${getCurrentCanvas().name}】吗？`,
     buttons: [
       { text: '取消', style: '', value: false },
       { text: '删除', style: 'destructive', value: true }
     ]
   });
   if (res.value) {
     canvases = canvases.filter(c => c.id !== activeCanvasId);
     activeCanvasId = canvases[0].id;
     highlightSelectedPiece(null);
     renderTabs();
     renderCurrentCanvasPieces();
   }
 }

 async function clearCurrentCanvas() {
   const currentCanvas = getCurrentCanvas();
   if (!currentCanvas || currentCanvas.pieces.length === 0) return;
   const res = await showIOSAlert({
     title: '清空画布',
     message: '确定要清空当前画布上的所有图形吗？',
     buttons: [
       { text: '取消', style: '', value: false },
       { text: '清空', style: 'destructive', value: true }
     ]
   });
   if (res.value) {
     currentCanvas.pieces = [];
     highlightSelectedPiece(null);
     renderCurrentCanvasPieces();
   }
 }

 function getCurrentCanvas() {
   return canvases.find(c => c.id === activeCanvasId);
 }

 function createPieceData(type, x, y) {
   const def = TANGRAM_DEFINITIONS.find(d => d.type === type);
   if (!def) return null;
   const id = 'piece-' + (++pieceIdCounter);

   let sumX = 0, sumY = 0;
   def.rawPoints.forEach(p => { sumX += p.x; sumY += p.y; });
   const cx = sumX / def.rawPoints.length;
   const cy = sumY / def.rawPoints.length;

   const localPoints = def.rawPoints.map(p => ({
     x: Number((p.x - cx).toFixed(4)),
     y: Number((p.y - cy).toFixed(4))
   }));

   let minX = Infinity, maxX = -Infinity, minY = Infinity, maxY = -Infinity;
   localPoints.forEach(p => {
     if (p.x < minX) minX = p.x;
     if (p.x > maxX) maxX = p.x;
     if (p.y < minY) minY = p.y;
     if (p.y > maxY) maxY = p.y;
   });

   return {
     id,
     type: def.type,
     name: def.name,
     color: def.color,
     rawPoints: localPoints,
     points: localPoints.map(p => `${p.x},${p.y}`).join(' '),
     bbox: { minX, maxX, minY, maxY, width: maxX - minX, height: maxY - minY },
     x: Math.round(x),
     y: Math.round(y),
     rotation: 0,
     scaleX: 1
   };
 }

 function renderCurrentCanvasPieces() {
   svgCanvas.innerHTML = '';
   const currentCanvas = getCurrentCanvas();
   if (!currentCanvas) return;
   currentCanvas.pieces.forEach(piece => renderPieceSVG(piece));
   if (selectedPieceId) highlightSelectedPiece(selectedPieceId);
 }

 function renderPieceSVG(piece) {
   const g = document.createElementNS('http://www.w3.org/2000/svg', 'g');
   g.setAttribute('id', piece.id);
   g.setAttribute('class', 'tangram-piece');
   updateTransform(g, piece);

   const polygon = document.createElementNS('http://www.w3.org/2000/svg', 'polygon');
   polygon.setAttribute('points', piece.points);
   polygon.setAttribute('fill', piece.color);
   polygon.setAttribute('stroke', '#475569');
   polygon.setAttribute('stroke-width', '1.5');
   polygon.setAttribute('stroke-linejoin', 'round');
   g.appendChild(polygon);

   const controlsG = document.createElementNS('http://www.w3.org/2000/svg', 'g');
   controlsG.setAttribute('class', 'selection-controls');
   controlsG.style.display = 'none';

   const padding = 10;
   const bMinX = piece.bbox.minX - padding;
   const bMaxX = piece.bbox.maxX + padding;
   const bMinY = piece.bbox.minY - padding;
   const bMaxY = piece.bbox.maxY + padding;
   const bWidth = bMaxX - bMinX;
   const bHeight = bMaxY - bMinY;

   const rect = document.createElementNS('http://www.w3.org/2000/svg', 'rect');
   rect.setAttribute('x', bMinX);
   rect.setAttribute('y', bMinY);
   rect.setAttribute('width', bWidth);
   rect.setAttribute('height', bHeight);
   rect.setAttribute('fill', 'none');
   rect.setAttribute('stroke', '#2563eb');
   rect.setAttribute('stroke-width', '1.2');
   rect.setAttribute('stroke-dasharray', '4 3');
   controlsG.appendChild(rect);

   const cornerSize = 7;
   const corners = [
     { x: bMinX - cornerSize / 2, y: bMinY - cornerSize / 2 },
     { x: bMaxX - cornerSize / 2, y: bMinY - cornerSize / 2 },
     { x: bMinX - cornerSize / 2, y: bMaxY - cornerSize / 2 },
     { x: bMaxX - cornerSize / 2, y: bMaxY - cornerSize / 2 }
   ];
   corners.forEach(c => {
     const sq = document.createElementNS('http://www.w3.org/2000/svg', 'rect');
     sq.setAttribute('x', c.x);
     sq.setAttribute('y', c.y);
     sq.setAttribute('width', cornerSize);
     sq.setAttribute('height', cornerSize);
     sq.setAttribute('fill', '#ffffff');
     sq.setAttribute('stroke', '#2563eb');
     sq.setAttribute('stroke-width', '1.5');
     controlsG.appendChild(sq);
   });

   const handleY = bMinY - 30;
   const line = document.createElementNS('http://www.w3.org/2000/svg', 'line');
   line.setAttribute('x1', 0);
   line.setAttribute('y1', bMinY);
   line.setAttribute('x2', 0);
   line.setAttribute('y2', handleY);
   line.setAttribute('stroke', '#2563eb');
   line.setAttribute('stroke-width', '1.2');
   controlsG.appendChild(line);

   const rotateCircle = document.createElementNS('http://www.w3.org/2000/svg', 'circle');
   rotateCircle.setAttribute('cx', 0);
   rotateCircle.setAttribute('cy', handleY);
   rotateCircle.setAttribute('r', 9);
   rotateCircle.setAttribute('fill', '#ffffff');
   rotateCircle.setAttribute('stroke', '#2563eb');
   rotateCircle.setAttribute('stroke-width', '2');
   rotateCircle.setAttribute('class', 'rotate-handle-circle');
   controlsG.appendChild(rotateCircle);

   if (piece.type === 'parallelogram') {
     const flipG = document.createElementNS('http://www.w3.org/2000/svg', 'g');
     flipG.setAttribute('class', 'flip-button-icon');
     flipG.setAttribute('transform', `translate(${bMaxX}, ${bMinY - 12})`);

     const flipCircle = document.createElementNS('http://www.w3.org/2000/svg', 'circle');
     flipCircle.setAttribute('cx', 0);
     flipCircle.setAttribute('cy', 0);
     flipCircle.setAttribute('r', 11);
     flipCircle.setAttribute('fill', '#ffffff');
     flipCircle.setAttribute('stroke', '#2563eb');
     flipCircle.setAttribute('stroke-width', '1.5');

     const flipPath = document.createElementNS('http://www.w3.org/2000/svg', 'path');
     flipPath.setAttribute('d', 'M-5,0 L5,0 M-2,-3 L-5,0 L-2,3 M2,-3 L5,0 L2,3');
     flipPath.setAttribute('stroke', '#2563eb');
     flipPath.setAttribute('stroke-width', '1.2');
     flipPath.setAttribute('fill', 'none');

     flipG.appendChild(flipCircle);
     flipG.appendChild(flipPath);
     controlsG.appendChild(flipG);

     flipG.addEventListener('pointerdown', (e) => {
       e.stopPropagation();
       piece.scaleX = (piece.scaleX || 1) * -1;
       updateTransform(g, piece);
       applyVertexSnapping(piece);
       updateTransform(g, piece);
     });
   }

   g.appendChild(controlsG);
   svgCanvas.appendChild(g);

   bindPieceEvents(g, piece, rotateCircle);
 }

 function updateTransform(el, piece) {
   const scaleX = piece.scaleX || 1;
   el.setAttribute('transform', `translate(${piece.x}, ${piece.y}) rotate(${piece.rotation}) scale(${scaleX}, 1)`);
 }

 function highlightSelectedPiece(id) {
   if (selectedPieceId && selectedPieceId !== id) {
     const prev = document.getElementById(selectedPieceId);
     if (prev) {
       const prevCtrl = prev.querySelector('.selection-controls');
       if (prevCtrl) prevCtrl.style.display = 'none';
     }
   }
   selectedPieceId = id;
   if (id) {
     const current = document.getElementById(id);
     if (current) {
       const currentCtrl = current.querySelector('.selection-controls');
       if (currentCtrl) currentCtrl.style.display = 'block';
       svgCanvas.appendChild(current);
     }
   }
 }

 // 计算标准角度磁吸吸附（0°, 45°, 90°, 135°...）
 function snapAngle(rawAngle) {
   let normalized = rawAngle % 360;
   if (normalized < 0) normalized += 360;
   
   const nearestStep = Math.round(normalized / ANGLE_SNAP_INTERVAL) * ANGLE_SNAP_INTERVAL;
   const diff = Math.abs(normalized - nearestStep);

   if (diff <= ANGLE_SNAP_THRESHOLD || Math.abs(diff - 360) <= ANGLE_SNAP_THRESHOLD) {
     return nearestStep % 360;
   }
   return rawAngle;
 }

 function getTransformedVertices(piece) {
   const rad = (piece.rotation * Math.PI) / 180;
   const cos = Math.cos(rad);
   const sin = Math.sin(rad);
   const sx = piece.scaleX || 1;

   return piece.rawPoints.map(p => {
     const localX = p.x * sx;
     const localY = p.y;
     const rx = localX * cos - localY * sin;
     const ry = localX * sin + localY * cos;
     return {
       x: piece.x + rx,
       y: piece.y + ry
     };
   });
 }

 function applyVertexSnapping(currentPiece) {
   const currentCanvas = getCurrentCanvas();
   if (!currentCanvas) return;

   const currentVertices = getTransformedVertices(currentPiece);
   const otherPieces = currentCanvas.pieces.filter(p => p.id !== currentPiece.id);

   let minDistance = VERTEX_SNAP_DISTANCE;
   let snapOffsetX = 0;
   let snapOffsetY = 0;
   let foundSnap = false;

   for (const otherPiece of otherPieces) {
     const targetVertices = getTransformedVertices(otherPiece);
     for (const cV of currentVertices) {
       for (const tV of targetVertices) {
         const dx = tV.x - cV.x;
         const dy = tV.y - cV.y;
         const dist = Math.hypot(dx, dy);

         if (dist < minDistance) {
           minDistance = dist;
           snapOffsetX = dx;
           snapOffsetY = dy;
           foundSnap = true;
         }
       }
     }
   }

   if (foundSnap) {
     currentPiece.x += snapOffsetX;
     currentPiece.y += snapOffsetY;
   }
 }

 function bindPieceEvents(groupEl, piece, rotateCircle) {
   let isDragging = false;
   let isRotating = false;
   let startX, startY, startPieceX, startPieceY;
   let startAngle = 0, startRotation = 0;
   let centerX = 0, centerY = 0;

   groupEl.addEventListener('pointerdown', (e) => {
     if (isRotating) return;
     e.stopPropagation();
     highlightSelectedPiece(piece.id);
     isDragging = true;
     groupEl.setPointerCapture(e.pointerId);
     startX = e.clientX;
     startY = e.clientY;
     startPieceX = piece.x;
     startPieceY = piece.y;
   });

   groupEl.addEventListener('pointermove', (e) => {
     if (isDragging) {
       const dx = e.clientX - startX;
       const dy = e.clientY - startY;
       piece.x = startPieceX + dx;
       piece.y = startPieceY + dy;
       updateTransform(groupEl, piece);
     }
   });

   const handlePointerUp = (e) => {
     if (isDragging) {
       applyVertexSnapping(piece);
       updateTransform(groupEl, piece);
       isDragging = false;
     }
   };

   groupEl.addEventListener('pointerup', handlePointerUp);
   groupEl.addEventListener('pointercancel', handlePointerUp);

   // 连续旋转 + 45° 视角卡位磁吸
   rotateCircle.addEventListener('pointerdown', (e) => {
     e.stopPropagation();
     highlightSelectedPiece(piece.id);
     isRotating = true;
     rotateCircle.setPointerCapture(e.pointerId);

     const rect = svgCanvas.getBoundingClientRect();
     centerX = rect.left + piece.x;
     centerY = rect.top + piece.y;

     startAngle = Math.atan2(e.clientY - centerY, e.clientX - centerX) * (180 / Math.PI);
     startRotation = piece.rotation;
   });

   rotateCircle.addEventListener('pointermove', (e) => {
     if (!isRotating) return;
     e.stopPropagation();

     const currentAngle = Math.atan2(e.clientY - centerY, e.clientX - centerX) * (180 / Math.PI);
     let angleDiff = currentAngle - startAngle;

     if (angleDiff > 180) angleDiff -= 360;
     if (angleDiff < -180) angleDiff += 360;

     const rawRotation = startRotation + angleDiff;
     // 实时进行 45° 角度吸附修正
     piece.rotation = snapAngle(rawRotation);
     updateTransform(groupEl, piece);
   });

   const endRotate = (e) => {
     if (isRotating) {
       e.stopPropagation();
       // 确保最终释放时精准吸附在最近的 45° 整数刻度上
       piece.rotation = snapAngle(piece.rotation);
       applyVertexSnapping(piece);
       updateTransform(groupEl, piece);
       isRotating = false;
     }
   };

   rotateCircle.addEventListener('pointerup', endRotate);
   rotateCircle.addEventListener('pointercancel', endRotate);
 }

 function setupEventListeners() {
   const container = document.getElementById('canvas-container');
   container.addEventListener('dragover', (e) => e.preventDefault());
   container.addEventListener('drop', (e) => {
     e.preventDefault();
     const type = e.dataTransfer.getData('text/plain');
     if (type) {
       const rect = svgCanvas.getBoundingClientRect();
       const x = e.clientX - rect.left;
       const y = e.clientY - rect.top;
       const pieceData = createPieceData(type, x, y);
       if (pieceData) {
         const currentCanvas = getCurrentCanvas();
         currentCanvas.pieces.push(pieceData);
         renderPieceSVG(pieceData);
         highlightSelectedPiece(pieceData.id);
       }
     }
   });

   svgCanvas.addEventListener('click', (e) => {
     if (e.target === svgCanvas) {
       highlightSelectedPiece(null);
     }
   });

   btnAddCanvas.addEventListener('click', createNewCanvas);
   document.getElementById('btn-clear-canvas').addEventListener('click', clearCurrentCanvas);
   document.getElementById('btn-delete-canvas').addEventListener('click', deleteCurrentCanvas);
   
   document.getElementById('btn-delete-piece').addEventListener('click', () => {
     if (!selectedPieceId) return;
     const el = document.getElementById(selectedPieceId);
     if (el) el.remove();
     const currentCanvas = getCurrentCanvas();
     currentCanvas.pieces = currentCanvas.pieces.filter(p => p.id !== selectedPieceId);
     highlightSelectedPiece(null);
   });

   document.getElementById('btn-toggle-sidebar').addEventListener('click', () => {
     sidebar.classList.toggle('collapsed');
   });

   window.addEventListener('keydown', (e) => {
     if (e.key === 'Delete' || e.key === 'Backspace') {
       if (e.target.tagName !== 'INPUT') {
         document.getElementById('btn-delete-piece').click();
       }
     }
   });
 }
